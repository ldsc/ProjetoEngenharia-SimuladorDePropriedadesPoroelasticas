
\chapter{Título do Apêndice}

\lhead{Apêndice \thechapter: Titulo do Apêndice}\rhead{\thepage}Descreve-se
neste apêndice ...
\begin{itemize}
\item Os anexos ou apêndices contém material auxiliar. Por exemplo, tabelas,
gráficos, resultados de experimentos, algoritmos, códigos e simulações.
\item Um apêndice pode incluir assuntos mais gerais (geral demais para estar
no núcleo do trabalho) ou mais específicos (detalhado demais para
estar no núcleo do trabalho).
\item Pode conter um artigo de auxílio fundamental ao trabalho. 
\item Pode conter artigos publicados.
\item {[}tudo aquilo que for importante para a tese mas não essencial, deve
ser colocado em apêndices{]}
\item {[}como exemplo, revisão de metodologias, técnicas, modelos matemáticos,
ítens desenvolvidos por terceiros{]}
\item {[}algoritmos e programas devem ser colocados no apêndice{]}
\item {[}imagens detalhadas de programas desenvolvidos devem ser colocados
no apêndice{]}
\end{itemize}
