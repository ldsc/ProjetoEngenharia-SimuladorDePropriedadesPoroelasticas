@startuml
' Título do Diagrama
title Diagrama de Atividade - Fluxo do Software

' Estilo visual
skinparam activity {
    StartColor #FF0000
    EndColor #FF0000
    BorderColor #A0A0A0
    BackgroundColor #FEFBDC
    ArrowColor #555555
}
skinparam activityDiamond {
    BackgroundColor #FFEF96
    BorderColor #A0A0A0
    ArrowColor #555555
}

start

' O laço "repeat" garante que o fluxo se repita até que o usuário decida sair.
repeat
    :Apresentar Menu Principal;

    ' Decisão principal: Processar dados ou Sair.
    if (Escolhe 'Processar Amostra'?) then (Sim)
        ' Bloco de Entrada de Dados
        if (Escolhe 'Carregar de Arquivo'?) then (Sim)
            :Executar processo "Carregar Arquivo";
        else (Não, 'Criar Amostra')
            :Executar processo "Criar Amostra Manualmente";
        endif

        ' Após a entrada de dados, o fluxo continua sequencialmente
        :Executar cálculos de propriedades poroelásticas;
        :Resultados calculados e disponíveis;

        ' Bloco de Análise e Relatórios
        if (Escolhe 'Gerar Gráficos'?) then (Sim)
            :Executar processo "Gerar Gráficos";
        else (Não, 'Mostrar Relatório')
            :Executar processo "Mostrar Relatórios";
        endif
        :Análise/Relatório concluído;

    else (Não, escolhe 'Sair')
        ' Se o usuário escolher sair no menu principal, esta condição
        ' fará com que o laço de repetição termine.
        stop
    endif

' A condição do laço é verificada aqui.
repeat while (Deseja realizar outra operação?) is (Sim)
->Não;

:Finalizar Software;

stop

@enduml
