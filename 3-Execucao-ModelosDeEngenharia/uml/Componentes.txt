@startuml
' Título do Diagrama
title Diagrama de Componentes

' Estilo visual e de leiaute
left to right direction
skinparam linetype ortho
skinparam componentStyle uml2

' --- Estilos para os Pacotes (Áreas) ---
skinparam package<<Interfaces>> {
    BackgroundColor #E6E6FA ' Lilás claro
    BorderColor #5D5D9E
}
skinparam package<<Implementações>> {
    BackgroundColor #FFF0F5 ' Rosa claro
    BorderColor #9E5D5D
}
skinparam package<<Artefatos>> {
    BackgroundColor #F0FFFF ' Azul claro
    BorderColor #5D9E9E
}
skinparam component {
    BackgroundColor #FFFFFF
    BorderColor #000000
}
skinparam artifact {
    BackgroundColor #EFEFEF
    BorderColor #333333
}


' --- Área 1: Interfaces (.h) ---
package "Interfaces (.h)" <<Interfaces>> {
    component "CPECApplication.h" as AppH <<arquivo>>
    component "CInput.h" as InputH <<arquivo>>
    component "CElasticCalculator.h" as CalcH <<arquivo>>
    component "CPlotManager.h" as PlotH <<arquivo>>
    component "CReservoirData.h" as ResDataH <<arquivo>>
    component "CMineralDatabase.h" as DBClassH <<arquivo>>
}

' --- Área 2: Implementações e Dependências do Código ---
package "Implementações (.cpp)" <<Implementações>> {
    component "main.cpp" as MainCPP <<arquivo>>
    component "CPECApplication.cpp" as AppCPP <<arquivo>>
    component "CInput.cpp" as InputCPP <<arquivo>>
    component "CElasticCalculator.cpp" as CalcCPP <<arquivo>>
    component "CPlotManager.cpp" as PlotCPP <<arquivo>>
    component "CReservoirData.cpp" as ResDataCPP <<arquivo>>
    component "CMineralDatabase.cpp" as DBClassCPP <<arquivo>>
}

' --- Área 3: Demais Componentes (Artefatos Finais) ---
package "Artefatos e Dependências de Execução" <<Artefatos>> {
    artifact "PEC.exe" as Executable <<executável>>
    artifact "Gnuplot" as Gnuplot <<ferramenta>>
    artifact "Minerais.csv" as MineralDB <<dados>>
    artifact "Amostras.txt" as SampleFile <<dados>>
}


' --- Relacionamentos entre as Áreas ---

' As implementações dependem de suas respectivas interfaces
MainCPP ..> AppH
AppCPP ..> AppH
InputCPP ..> InputH
CalcCPP ..> CalcH
PlotCPP ..> PlotH
ResDataCPP ..> ResDataH
DBClassCPP ..> DBClassH

' O pacote de implementação gera o executável
package "Implementações (.cpp)" ..> Executable : <<compila de>>

' O executável depende dos outros artefatos para rodar
Executable ..> Gnuplot
Executable ..> MineralDB
Executable ..> SampleFile

@enduml
