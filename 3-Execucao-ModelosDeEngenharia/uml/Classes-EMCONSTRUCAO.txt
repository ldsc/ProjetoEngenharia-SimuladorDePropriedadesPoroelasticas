@startuml

title Diagrama de Classes - Estrutura Completa

' --- Definição das Classes ---
class CSimulador {
  -m_input: CInput
  -m_database: CMineralDatabase
  -m_calculator: CElasticCalculator
  -m_reservoirData: CReservoirData
  -m_plotManager: CPlotManager
  +CSimulador()
  +~CSimulador()
  +Run()
  -Initialize()
  -Menu()
  -ProcessSamples()
  -Finalize()
}

class CInput {
  +CInput()
  +~CInput()
  +PromptManualInput(): SampleInput
  +ReadFromFile(filePath): SampleInput
  +ReadMultipleFromFile(filePath): vector<SampleInput>
  -ParseCSVLine(line, mineralsOut)
}

class CPlotManager {
  +CPlotManager()
  +PlotDepthProfile(data)
  +PlotPorosityProfile(data)
  +PlotVariationByMineral(db, principal, secundario)
}

class CElasticCalculator {
  -m_database: CMineralDatabase&
  +CElasticCalculator(db)
  +Calculate(input): Result
  -CalcVoigt(minerals, bulk): float
  -CalcReuss(minerals, bulk): float
}

class CShearCalculator {
  -m_database: CMineralDatabase&
  +CShearCalculator(db)
  +CalcVoigt(minerals, bulk): float
  +CalcReuss(minerals, bulk): float
}

class CReservoirData {
  -m_samples: vector<EnrichedSample>
  +CReservoirData()
  +AddSample(input, result)
  +GetSamples(): const vector<EnrichedSample>&
  +PrintSummary()
  +GetKGVvsDepth(depths, bulk, shear)
  +GetKGVvsPorosity(porosities, bulk, shear)
  +GetKGVvsDepthFull(depths, kB, kV, kR, kH, gB, gV, gR, gH)
  +GetKGVvsPorosityFull(porosities, kB, kV, kR, kH, gB, gV, gR, gH)
}

class CMineralDatabase {
  -m_minerals: unordered_map
  -new_attribute: undef
  -m_databaseFilePath
  +CMineralDatabase(filePath)
  +HasMineral(name): bool
  +GetBulkModulus(name): float
  +GetShearModulus(name): float
  +AddMineral(name, k, g)
  +SaveToFile(filePath): bool
  +PrintAllMineralsSorted()
  -LoadFromFile(filePath)
}

class CBulkCalculator {
  -m_database: CMineralDatabase
  +CBulkCalculator(db)
  -CalcVoigt(minerals, bulk): float
  -CalcReuss(minerals, bulk): float
}

class CMineral {
  -name
  -concentration
}

class CAmostra {
  -m_bulk
  -m_shear
}
