@startuml
autonumber

actor Usuário as U

participant "c:CInput" as CI <<Controlador>>
participant "s:CCalcSimulador" as CCS <<Controlador Principal>>
participant "m:CAmostra" as CM <<Entidade Amostra>>
participant "d:CMineralsDatabase" as CMD <<Base de Dados>>
participant "r:CReservoirData" as CRD <<Dados do Reservatório>>

' --- Fluxo: Cria Amostra ---

U -> CI: IniciarCriaAmostra()
activate CI

CI -> U: SolicitarParametrosAmostra()

U -> CI: InserirParametrosAmostra(nome, tipo, ...)
CI -> CI: ValidarParametros()
activate CI
deactivate CI

alt Se parâmetros válidos
    CI -> CM: <<create>> CriarAmostra(parametros)
    activate CM
    CM -> CMD: ObterDadosIniciais(parametros)
    activate CMD
    CMD --> CM: DadosMineraisIniciais
    deactivate CMD
    CM -> CRD: ObterPropriedadesDefault(tipoAmostra)
    activate CRD
    CRD --> CM: PropriedadesReservatorioDefault
    deactivate CRD
    CM --> CI: NovaAmostraCriada(m)
    deactivate CM
    
    CI -> CCS: RegistrarAmostra(m)
    activate CCS
    CCS -> CMD: SalvarAmostra(m)
    activate CMD
    CMD --> CCS: AmostraSalva
    deactivate CMD
    CCS --> CI: AmostraRegistrada
    deactivate CCS

    CI --> U: Amostra Criada com Sucesso: {nome}
else Se parâmetros inválidos
    CI --> U: Erro: Parâmetros de amostra inválidos.
end

deactivate CI

@enduml
