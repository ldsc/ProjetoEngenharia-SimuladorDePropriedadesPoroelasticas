@startuml
title Diagrama de Caso de Uso - Carregar Arquivo
left to right direction

skinparam rectangle {
    BorderColor #000000
    BackgroundColor #FFFFFF
}
skinparam usecase {
    BackgroundColor #FEFBDC
    BorderColor #000000
}
skinparam database {
    BackgroundColor #EFEFEF
    BorderColor #333333
}
skinparam artifact {
  BackgroundColor #EFEFEF
  BorderColor #333333
}

' Ator Principal
actor Usuário

' Sistema e Casos de Uso
rectangle "PoreElasticCalculator" {
  usecase "Carregar Arquivo" as UC_CarregarArquivo

  ' Casos de Uso incluídos (sub-funcionalidades necessárias)
  usecase "Selecionar Arquivo" as UC_SelecionarArquivo
  usecase "Validar Conteúdo do Arquivo" as UC_ValidarConteudo
  usecase "Importar Dados do Arquivo" as UC_ImportarDados
  usecase "Exibir Confirmação de Carregamento" as UC_Confirmacao

  ' Casos de Uso estendidos (comportamentos opcionais/exceções)
  usecase "Informar Erro de Formato Inválido" as UC_ErroFormato
  usecase "Informar Erro de Leitura/Importação" as UC_ErroLeitura
}

' Entidades Externas
artifact "Arquivo de Dados\n(.txt, .csv, .drx)" as DataFile
database "Banco de Dados\nde Minerais" as DB

' Relacionamentos
Usuário -- UC_CarregarArquivo

UC_CarregarArquivo .> UC_SelecionarArquivo : <<include>>
UC_CarregarArquivo .> UC_ValidarConteudo : <<include>>
UC_CarregarArquivo .> UC_ImportarDados : <<include>>
UC_CarregarArquivo .> UC_Confirmacao : <<include>>

UC_ValidarConteudo <. UC_ErroFormato : <<extend>>
UC_ImportarDados <. UC_ErroLeitura : <<extend>>

UC_CarregarArquivo ..> DataFile : "lê"
UC_ImportarDados ..> DB : "salva"

@enduml
